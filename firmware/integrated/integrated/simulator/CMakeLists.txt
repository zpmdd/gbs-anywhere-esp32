cmake_minimum_required(VERSION 3.16)
project(gbs_touch_simulator C)
set(CMAKE_C_STANDARD 11)
option(HUB_AMOLED_216 "Build the 480 x 480 AMOLED control surface" OFF)
include(FetchContent)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../display/lv_conf.h" CACHE STRING "" FORCE)
FetchContent_Declare(lvgl URL https://github.com/lvgl/lvgl/archive/refs/tags/v8.4.0.tar.gz)
FetchContent_MakeAvailable(lvgl)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
if(NOT DEFINED ENV{IDF_PATH})
    message(FATAL_ERROR "Source ESP-IDF export.sh to reuse its cJSON source")
endif()
set(JSON_DIR "$ENV{IDF_PATH}/components/json/cJSON")
add_library(sim_json STATIC "${JSON_DIR}/cJSON.c")
target_include_directories(sim_json PUBLIC "${JSON_DIR}")
file(GLOB display_fonts CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../display/fonts/*.c")
add_executable(gbs_touch_simulator main.c ../display/gbs_ui.c ../display/gbs_ui_model.c
    ../display/icons/gbs_icons.c ${display_fonts})
target_include_directories(gbs_touch_simulator PRIVATE ../display)
if(HUB_AMOLED_216)
    target_compile_definitions(gbs_touch_simulator PRIVATE HUB_AMOLED_216=1)
endif()
target_link_libraries(gbs_touch_simulator PRIVATE lvgl PkgConfig::SDL2 sim_json m)
target_compile_options(gbs_touch_simulator PRIVATE -Wall -Wextra -Werror)
if(APPLE)
    enable_language(OBJC)
    target_sources(gbs_touch_simulator PRIVATE cocoa_input.m)
    target_link_libraries(gbs_touch_simulator PRIVATE "-framework Cocoa")
endif()
enable_testing()
add_test(NAME native_ui COMMAND gbs_touch_simulator --self-test)
set_tests_properties(native_ui PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
if(HUB_AMOLED_216)
    add_executable(test_device_input_lvgl ../tests/test_device_input_lvgl.c ../display/gbs_device_policy.c)
    target_include_directories(test_device_input_lvgl PRIVATE ../display)
    target_link_libraries(test_device_input_lvgl PRIVATE lvgl m)
    target_compile_options(test_device_input_lvgl PRIVATE -Wall -Wextra -Werror)
    add_test(NAME device_input_lvgl COMMAND test_device_input_lvgl)
endif()
